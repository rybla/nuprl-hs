-- | Foundational list theory: formation, constructors, discrimination,
-- the structural recursor, and algebraic reduction laws.
--
-- This theory formalizes polymorphic list operations (Append, Length, Map,
-- IsEmpty), establishes their definitional reduction equations at nil and cons,
-- proves constructor discrimination (`nil_ne_cons`), and provides exemplary
-- evaluations for practical programming languages theory work.
theory Lists

abs Append(xs; ys) == list_ind xs (ys) (h, t, ih. h :: ih)
abs Length(xs) == list_ind xs 0 (h, t, ih. ih + 1)
abs Map(f; xs) == list_ind xs [] (h, t, ih. f h :: ih)
abs IsEmpty(xs) == list_ind xs True (h, t, ih. False)

soft Append, Length, Map, IsEmpty
