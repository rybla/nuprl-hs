-- | Lists: nil, cons, the recursor, and closed computation of `list_ind`.
--
-- A singleton list is `h :: []`; `[t]` is squash, `[a, b, …]` (with a
-- comma) is a list literal.
theory Lists

abs Append(xs; ys) == list_ind xs (ys) (h, t, ih. h :: ih)
abs Length(xs) == list_ind xs 0 (h, t, ih. ih + 1)

soft Append, Length

proof
  intro A

    hyp 3,

  ]
qed
