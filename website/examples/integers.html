-- | Integers: closed arithmetic, comparison, decidability, and induction.
--
-- Open terms do not reduce (`n + 0` stays stuck), so identities such as
-- `n + 0 = n` are not discharged by `arith`. Closed numerals compute.
theory Integers

  intro x

  decide x < y THENL [left THEN hyp, right THEN hyp]
qed

theorem decide_closed_eq :
  2 = 2 ∈ Int ∨ ¬(2 = 2 ∈ Int)
proof
  decide 2 = 2 THENL [left THEN hyp, right THEN hyp]
qed
