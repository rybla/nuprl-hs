Integers

Discrete decidability, strict order decidability, sign splitting, and ternary integer induction.

Integers: structural decidability, order, induction, and exemplary arithmetic.

Open integer arithmetic does not reduce symbolically, but closed numerals compute definitionally. This theory formalizes the discrete decidability of integer equality and strict order (<), sign splitting, the ternary induction principle (`ind`), and minimal exemplary evaluations.

-- | Integers: structural decidability, order, induction, and exemplary arithmetic.
--
-- Open integer arithmetic does not reduce symbolically, but closed numerals
-- compute definitionally. This theory formalizes the discrete decidability
-- of integer equality and strict order (<), sign splitting, the ternary
-- induction principle (`ind`), and minimal exemplary evaluations.
theory Integers

abs Discrete(T) == ∀x:T. ∀y:T. x = y ∈ T ∨ ¬(x = y ∈ T)
abs DecidableLt == ∀x:Int. ∀y:Int. x < y ∨ ¬(x < y)

soft Discrete, DecidableLt

--------------------------------------------------------------------------------
-- Discreteness and decidable order
--------------------------------------------------------------------------------

theorem int_discrete :
  Discrete(Int)
proof
  intro x
  intro y
  decide x = y THENL [left THEN hyp, right THEN hyp]
qed

theorem int_lt_decidable :
  DecidableLt
proof
  intro x
  intro y
  decide x < y THENL [left THEN hyp, right THEN hyp]
qed

theorem sign_split :
  ∀x:Int. (x = 0 ∈ Int) ∨ ¬(x = 0 ∈ Int)
proof
  intro x
  decide x = 0 THENL [left THEN hyp, right THEN hyp]
qed

--------------------------------------------------------------------------------
-- Order properties and decision combinators
--------------------------------------------------------------------------------

theorem not_one_lt_zero :
  ¬(1 < 0)
proof
  intro p
  elim 1
qed

theorem zero_le_one :
  0 ≤ 1
proof
  intro
qed

theorem int_eq_same :
  int_eq 4 4 1 0 = 1 ∈ Int
proof
  eq
qed

theorem less_true :
  less 0 1 7 9 = 7 ∈ Int
proof
  eq
qed

--------------------------------------------------------------------------------
-- Induction on integers (`ind` / `elim` on Int)
--------------------------------------------------------------------------------

-- Counting upward with the integer recursor
theorem ind_counts :
  ind(4; x,ih. 0; 0; k,r. r + 1) = 4 ∈ Int
proof
  eq
qed

-- 3-branch induction on integers: base (0), successor (>0), predecessor (<0)
theorem int_to_unit :
  ∀n:Int. Unit
proof
  intro n
  elim 1 THENL [intro, intro, intro]
qed

theorem int_eq_refl :
  ∀n:Int. n = n ∈ Int
proof
  intro n
  hyp
qed

theorem zero_ne_one :
  ¬(0 = 1 ∈ Int)
proof
  intro p
  elim 1
qed

--------------------------------------------------------------------------------
-- Exemplary arithmetic evaluations
--------------------------------------------------------------------------------

theorem one_plus_one :
  1 + 1 = 2 ∈ Int
proof
  eq
qed

theorem arith_chain :
  (1 + 2) * 3 = 9 ∈ Int
proof
  arith
qed
