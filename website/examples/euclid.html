theory Euclid

Euclidean algorithm

Iterated remainder computes gcd on closed integers, with divisibility proofs.

Verified Euclidean algorithm and divisibility theory.

`GcdStep` is one remainder step `<a, b> ↦ <b, a % b>` (or the identity when `b = 0`). `Euclid(a; b)` iterates that step a fixed number of times (8 is enough for typical small instances) and returns the first component. Closed instances compute, and the results are shown to satisfy the specification of common divisors and coprimality.

-- | Verified Euclidean algorithm and divisibility theory.
--
-- `GcdStep` is one remainder step `<a, b> ↦ <b, a % b>` (or the identity
-- when `b = 0`). `Euclid(a; b)` iterates that step a fixed number of times
-- (8 is enough for typical small instances) and returns the first
-- component. Closed instances compute, and the results are shown to satisfy
-- the specification of common divisors and coprimality.
theory Euclid

abs Fst(p) == let <u, v> = p in u
abs Snd(p) == let <u, v> = p in v
abs GcdStep(p) ==
  let <a, b> = p in
    int_eq b 0 p <b, a % b>
abs Euclid(a; b) ==
  Fst(ind(8; x,ih. <a, b>; <a, b>; k,r. GcdStep(r)))
abs Divides(d; n) == ∃k:Int. n = d * k ∈ Int
abs IsCommonDivisor(d; a; b) == Divides(d; a) ∧ Divides(d; b)
abs Coprime(a; b) == Euclid(a; b) = 1 ∈ Int

soft Fst, Snd, GcdStep, Euclid, Divides, IsCommonDivisor, Coprime

--------------------------------------------------------------------------------
-- Structural properties of the step and base cases
--------------------------------------------------------------------------------

-- When the remainder is zero, the step is the identity for any integer a.
theorem gcdstep_done :
  ∀a:Int. GcdStep(<a, 0>) = <a, 0> ∈ Int × Int
proof
  intro a
  eq THEN auto
qed

-- When the second argument is 0, Euclid terminates immediately returning a.
theorem euclid_zero_r :
  ∀a:Int. Euclid(a; 0) = a ∈ Int
proof
  intro a
  eq THEN auto
qed

--------------------------------------------------------------------------------
-- Concrete execution and common divisor verification
--------------------------------------------------------------------------------

theorem euclid_48_18 :
  Euclid(48; 18) = 6 ∈ Int
proof
  eq
qed

theorem euclid_48_18_common_divisor :
  IsCommonDivisor(Euclid(48; 18); 48; 18)
proof
  split THENL [
    exists 8 THENL [eq, eq],
    exists 3 THENL [eq, eq]
  ]
qed

--------------------------------------------------------------------------------
-- Coprimality
--------------------------------------------------------------------------------

theorem coprime_17_5 :
  Coprime(17; 5)
proof
  eq
qed

theorem not_coprime_48_18 :
  ¬Coprime(48; 18)
proof
  intro p
  elim 1
qed
